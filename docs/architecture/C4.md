```mermaid
  Person(eng, "Engineer / CI", "Runs evaluations, reads the gate")

  System_Ext(sut, "System under test", "fde-02 document assistant, or any /api/v1/ask service")
  System_Ext(llm, "LLM provider", "Mistral for the LLM judge and prompt-only baseline")
  System_Ext(src, "Dataset sources", "GitHub: FinanceBench, PubMedQA, CUAD (pinned commits)")

  Rel(sponsor, eval, "Dashboards, memo")
  Rel(eval, sut, "POST /api/v1/ask + traceparent", "HTTP")
  Rel(eval, llm, "Judge prompts", "HTTPS")
  Rel(eval, src, "Download + checksum", "HTTPS")
```
